# Verifica dell'esercizio modulo6/03_debounce
# I rimbalzi si simulano fermando il tempo virtuale (pause), dando piu' comandi e
# ripartendo (start): tutti i fronti arrivano nello stesso istante simulato.
attendi: Booting Zephyr OS
attendi: Pronto: premi sw0

# Pressione con rimbalzi: deve contare UNA volta
monitor: pause
monitor: sysbus.gpioPortA.sw0 Press
monitor: sysbus.gpioPortA.sw0 Release
monitor: sysbus.gpioPortA.sw0 Press
monitor: start
attendi: ^Pressioni valide: 1$
controlla: sysbus.gpioPortD.led0 State => True

# Rilascio con rimbalzi
monitor: pause
monitor: sysbus.gpioPortA.sw0 Release
monitor: sysbus.gpioPortA.sw0 Press
monitor: sysbus.gpioPortA.sw0 Release
monitor: start
attendi: ^Rilasciato

# Seconda pressione, pulita
monitor: sysbus.gpioPortA.sw0 Press
attendi: ^Pressioni valide: 2$
controlla: sysbus.gpioPortD.led0 State => False
monitor: sysbus.gpioPortA.sw0 Release
attendi: ^Rilasciato
pausa: 300
mai: Pressioni valide: 3
mai: Errore
