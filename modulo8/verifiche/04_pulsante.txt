# Verifica del passo 4 (conferma dell'allarme con SW0)
# Uso: .\verifica.ps1 modulo8\progetto -File modulo8\verifiche\04_pulsante.txt
# Il pulsante resta premuto finche' il firmware risponde: il debounce per
# campionamento (50 ms) ha sempre il tempo di vedere la pressione.
monitor: sysbus.twi0.bme280 Temperature 22
attendi: NODO AVVIATO
# Pressione senza allarme: non deve succedere nulla
monitor: sysbus.gpio0.sw0 Press
pausa: 500
monitor: sysbus.gpio0.sw0 Release
pausa: 200
monitor: sysbus.twi0.bme280 Temperature 35
attendi 30: ALLARME ATTIVO
controlla: sysbus.gpio0.led0 State => True
monitor: sysbus.gpio0.sw0 Press
attendi 5: ALLARME CONFERMATO
monitor: sysbus.gpio0.sw0 Release
controlla: sysbus.gpio0.led0 State => False
monitor: sysbus.twi0.bme280 Temperature 20
attendi 30: ALLARME RIENTRATO
